def load_price_config_from_cookie(cookie_file="cookie.xlsx"):
    """从cookie文件读取核价表"""
    if not os.path.exists(cookie_file):
        print(f"[错误] Cookie文件 {cookie_file} 不存在")
        return []

    try:
        # 读取Excel文件
        excel_file = pd.ExcelFile(cookie_file)
        price_df = None

        # 查找核价表工作表
        for sheet in excel_file.sheet_names:
            s = sheet.lower()
            if "核价" in s or "price" in s:
                print(f"[核价表] 找到【核价表】工作表：{sheet}")
                price_df = pd.read_excel(cookie_file, sheet_name=sheet)
                break

        if price_df is None:
            print("[错误] 未找到核价表工作表")
            return []

        # 转换为核价规则格式
        price_config = []
        for _, row in price_df.iterrows():
            start_with = str(row.get("startWith", "")).strip()
            red_line_price = row.get("redLinePrice", 0)
            if start_with and red_line_price > 0:
                price_config.append({
                    "startWith": start_with,
                    "redLinePrice": red_line_price
                })

        print(f"[核价表] 加载完成，共 {len(price_config)} 条核价规则")
        return price_config
    except Exception as e:
        print(f"[错误] 读取核价表失败: {str(e)}")
        return []


# ---------------------- 3. 限流控制工具类（优化版） ----------------------
class RateLimiter:
    def __init__(self, max_calls, period):
        self.max_calls = max_calls  # 周期内最大请求数
        self.period = period  # 周期（秒）
        self.calls = []
        self.lock = Semaphore(1)

    def wait(self):
        now = time.time()
        with self.lock:
            # 移除过期的请求记录
            self.calls = [t for t in self.calls if t > now - self.period]
            if len(self.calls) >= self.max_calls:
                # 等待到下一个请求窗口
                sleep_time = max(0.1, self.period - (now - self.calls[0]))
                time.sleep(sleep_time)
            self.calls.append(time.time())

def parse_cookies(cookie_str):
    cookies = {}
    for item in cookie_str.split(';'):
        if '=' in item:
            key, value = item.split('=', 1)
            cookies[key.strip()] = value.strip()
    return cookies

def batch_submit_review_result(submit_items, cookies):
    """批量提交审核结果（一次最多50条）- 单线程（核心修复：提交结果解析）"""
    if not submit_items:
        return {}

    # 限流控制
    RATE_LIMITER.wait()

    url = "https://agentseller.temu.com/api/kiana/magnus/mms/price/bargain-no-bom/batch"
    submit_result = {}  # key: spu_id, value: (success, msg)

    max_retry = CONFIG["pass_retry"] if any(item['result_code'] == 1 for item in submit_items) else 2

    for retry in range(max_retry):
        try:
            # 构造批量提交数据
            item_requests = []
            for item in submit_items:
                spu_id = item['spu_id']
                result_code = item['result_code']
                sku_ids = item['sku_ids']
                sku_price = item.get('sku_price')

                # 确保ID为整数类型
                try:
                    spu_id_int = int(spu_id)
                except:
                    spu_id_int = spu_id

                sku_ids_int = []
                for sku_id in sku_ids:
                    try:
                        sku_ids_int.append(int(sku_id))
                    except:
                        sku_ids_int.append(sku_id)

                request_item = {
                    "priceOrderId": spu_id_int,
                    "supplierResult": result_code,
                    "items": [{"productSkuId": sku_id, "price": sku_price} if result_code == 1
                              else {"productSkuId": sku_id}
                              for sku_id in sku_ids_int]
                }
                if result_code == 1:
                    request_item["confirmPrice"] = True
                item_requests.append(request_item)

            submit_data = {"itemRequests": item_requests}
            data = json.dumps(submit_data, separators=(',', ':'), ensure_ascii=False)
            print(f"[批量提交请求] 数据: {data[:300]}...")

            response = requests.post(
                url,
                headers=HEADERS,
                cookies=cookies,
                data=data.encode('utf-8'),
                timeout=30
            )

            print(f"\n[批量提交响应] 状态码: {response.status_code}")
            print(f"[批量提交响应] 内容: {response.text[:1000]}...")

            if response.status_code == 200:
                resp_content = json.loads(response.text)
                batch_result = resp_content.get('result', {}).get('batchOperateResult', {})
                top_success = resp_content.get('result', {}).get('success', False)
                global_error_msg = resp_content.get('errorMsg', '未知错误')

                # 核心修复1：兼容SPU ID的整数/字符串键（双向匹配）
                for item in submit_items:
                    spu_id = item['spu_id']
                    spu_success = False
                    error_msg = global_error_msg

                    # 尝试两种键类型匹配（先整数，后字符串）
                    spu_key_int = int(spu_id) if str(spu_id).isdigit() else None
                    spu_res = None

                    if spu_key_int is not None and spu_key_int in batch_result:
                        spu_res = batch_result[spu_key_int]
                    elif spu_id in batch_result:
                        spu_res = batch_result[spu_id]

                    # 解析单个SPU的结果
                    if spu_res:
                        spu_success = spu_res.get('success', False)
                        error_msg = spu_res.get('failDesc', '') if not spu_success else '提交成功'
                    else:
                        # 未找到该SPU的结果，但顶层成功，视为提交成功
                        if top_success:
                            spu_success = True
                            error_msg = '提交成功（响应中未单独返回该SPU结果）'
                        else:
                            spu_success = False
                            error_msg = '未获取到该SPU的提交结果'

                    submit_result[spu_id] = (spu_success, error_msg)

                # 核心修复2：精准统计成功数
                success_count = len([r for r in submit_result.values() if r[0]])
                print(
                    f"[批量提交结果] 共{len(submit_result)}条，成功{success_count}条，失败{len(submit_result) - success_count}条")
                return submit_result
            else:
                print(f"[批量提交异常] 状态码: {response.status_code}")
                submit_result = {
                    item['spu_id']: (False, f"状态码异常: {response.status_code}，响应内容: {response.text[:200]}")
                    for item in submit_items
                }
                return submit_result

        except Exception as e:
            error_msg = f"提交异常: {str(e)}"
            print(error_msg)
            traceback.print_exc()
            if retry < max_retry - 1:
                sleep_time = CONFIG["retry_delay"] * (retry + 1) * 2
                print(f"{sleep_time}秒后重试...")
                time.sleep(sleep_time)
            else:
                submit_result = {item['spu_id']: (False, error_msg) for item in submit_items}
                return submit_result

def match_red_line_price(code):
    """
    实现「最长startWith优先，不行降级到较短」的匹配逻辑
    步骤：1. 提取所有startWith并按长度降序排序 2. 逐个尝试匹配 3. 匹配到立即返回（不继续降级）
    """
    if not code:  # 空编码直接返回None
        return None

    # 步骤1：提取所有配置中的startWith，去重并按「长度降序」排序（最长的优先）
    # 生成 (startWith, 对应配置项) 的列表，便于后续匹配和取值
    start_with_list = []
    for cfg in PRICE_CONFIG:
        sw = cfg.get('startWith', '')
        if sw not in [item[0] for item in start_with_list]:  # 去重，避免重复匹配相同startWith
            start_with_list.append((sw, cfg))

    # 按startWith长度降序排序（核心：最长的排在前面，优先匹配）
    start_with_list.sort(key=lambda x: len(x[0]), reverse=True)

    # 步骤2：逐级降级匹配（最长→较短→最短）
    for sw, cfg in start_with_list:
        if len(sw) == 0:  # 跳过空字符串的startWith（无匹配意义）
            continue
        if code.startswith(sw):  # 找到匹配项，立即返回（实现「最长优先，匹配即停止」）
            print(f"[匹配成功（优先级{len(sw)}）] 编码{code} → 匹配{sw}（长度{len(sw)}）→ 红线价{cfg['redLinePrice']}")
            return cfg['redLinePrice']

    # 步骤3：所有startWith都匹配失败
    print(f"[匹配失败] 编码{code}无任何可匹配的startWith配置")
    write_unmatched_product(code)
    return None


def collect_products_to_process(data_list, page_num):
    """收集当前页所有需要处理的商品信息（核心修复：提取SKU层级的extCode作为有效code）"""
    products_to_process = []
    spu_id_set = set()

    print(f"\n===== 收集第{page_num}页待处理商品 =====")

    for idx, item in enumerate(data_list, 1):
        try:
            for skc in item.get('skcList', []):
                for review in skc.get('supplierPriceReviewInfoList', []):
                    # 基础过滤：有效状态 + 处理次数未超限
                    if review.get('status') == 1 and review.get("times", 0) < 5:
                        # 提取核心字段
                        spu_id = str(review.get('priceOrderId', ''))  # 统一转为字符串（原始SPU ID）
                        product_sku_list = review.get('productSkuList', [])
                        sku_ids = [str(sku.get('skuId', '')) for sku in product_sku_list if sku.get('skuId')]

                        # 核心修复：遍历SKU提取extCode（去重，取有效非空值）
                        code = ""
                        sku_code_set = set()
                        for sku in product_sku_list:
                            sku_ext_code = sku.get('extCode', '').strip()
                            if sku_ext_code:
                                sku_code_set.add(sku_ext_code)
                        # 取第一个有效code（所有SKU的code通常一致，若不一致取唯一值）
                        if sku_code_set:
                            code = list(sku_code_set)[0]

                        # 过滤条件：信息完整（spu_id/sku_ids/code均非空） + SPU未重复
                        if all([spu_id, sku_ids, code]) and spu_id not in spu_id_set:
                            spu_id_set.add(spu_id)

                            # 匹配红线价
                            red_line_price = match_red_line_price(code)
                            if red_line_price is None:
                                print(f"[过滤] 编码{code}无匹配红线价，跳过SPU: {spu_id}")
                                continue

                            # 组装待处理商品信息
                            products_to_process.append({
                                'spu_id': spu_id,
                                'sku_ids': sku_ids,
                                'code': code,
                                'red_line_price': red_line_price,
                                'idx': idx,
                                'page_num': page_num
                            })
                            print(f"[收集成功] SPU: {spu_id}（编码：{code}）已纳入待处理列表")
                        else:
                            # 打印过滤详情（便于调试）
                            empty_fields = []
                            if not spu_id: empty_fields.append("spu_id")
                            if not sku_ids: empty_fields.append("sku_ids")
                            if not code: empty_fields.append("code")
                            if spu_id in spu_id_set: empty_fields.append("重复SPU")
                            if empty_fields:
                                print(f"[过滤] 第{page_num}页第{idx}个商品 - 原因：{','.join(empty_fields)}")

        except Exception as e:
            print(f"[收集异常] 第{page_num}页第{idx}个商品: {str(e)}")
            traceback.print_exc()

    print(f"第{page_num}页收集完成：共 {len(products_to_process)} 条待处理商品")
    return products_to_process


def process_batch_products(products_to_process, cookies):
    """单线程批量处理商品：先批量获取价格，再批量提交结果"""
    if not products_to_process:
        return

    # 步骤1：按批量大小拆分，批量获取价格
    batch_size = CONFIG["batch_size"]
    spu_batches = [products_to_process[i:i + batch_size] for i in range(0, len(products_to_process), batch_size)]
    all_price_result = {}

    for batch_idx, spu_batch in enumerate(spu_batches, 1):
        print(f"\n----- 批量获取价格（第{batch_idx}/{len(spu_batches)}批，{len(spu_batch)}条SPU） -----")
        spu_ids = [p['spu_id'] for p in spu_batch]
        price_result = batch_get_suggest_price(spu_ids, cookies)
        all_price_result.update(price_result)
        time.sleep(CONFIG["request_delay"] / 2)  # 批次间短间隔

    # 步骤2：筛选有效商品并准备提交数据
    submit_items = []
    for product in products_to_process:
        spu_id = product['spu_id']
        suggest_price = all_price_result.get(spu_id)

        # 过滤无效价格
        if suggest_price is None:
            print(f"[过滤] SPU {spu_id} 无有效建议价，跳过")
            continue

        # 价格判断
        red_line_price = product['red_line_price']
        if CONFIG["use_price_check"]:
            if suggest_price >= red_line_price:
                result_code = 1
                decision = "通过"
            else:
                result_code = 3
                decision = "不通过"
            print(
                f"[价格判断] SPU {spu_id} - 建议价{suggest_price} {'≥' if result_code == 1 else '<'} 红线价{red_line_price} → {decision}")
        else:
            result_code = CONFIG["force_result"]
            decision = "强制通过" if result_code == 1 else "强制不通过"
            print(f"[强制结果] SPU {spu_id} → {decision}")

        # 准备提交数据
        if CONFIG["submit_enabled"]:
            submit_item = {
                'spu_id': spu_id,
                'sku_ids': product['sku_ids'],
                'result_code': result_code,
                'code': product['code']
            }
            if result_code == 1:
                submit_item['sku_price'] = suggest_price
            submit_items.append(submit_item)
        else:
            print(f"[预览] SPU {spu_id} - 提交功能已禁用")

    # 步骤3：批量提交结果
    if CONFIG["submit_enabled"] and submit_items:
        submit_batches = [submit_items[i:i + batch_size] for i in range(0, len(submit_items), batch_size)]
        for batch_idx, submit_batch in enumerate(submit_batches, 1):
            print(f"\n----- 批量提交结果（第{batch_idx}/{len(submit_batches)}批，{len(submit_batch)}条SPU） -----")
            submit_result = batch_submit_review_result(submit_batch, cookies)

            # 输出提交详情
            for item in submit_batch:
                spu_id = item['spu_id']
                success, msg = submit_result.get(spu_id, (False, "未获取结果"))
                print(f"[提交详情] SPU {spu_id} - {'✅ 成功' if success else '❌ 失败'}: {msg}")

            time.sleep(CONFIG["request_delay"] / 2)

            print(f"[页面结果] 第{page_num}页 - 商品数量: {len(data_list)}")
        except json.JSONDecodeError:
            print(f"[页面异常] 第{page_num}页数据解析失败，内容: {response.text[:100]}，停止处理")
            break

        if not data_list:
            print(f"[页面结果] 第{page_num}页无数据，结束处理")
            break

        # 收集并批量处理当前页商品
        products_to_process = collect_products_to_process(data_list, page_num)
        process_batch_products(products_to_process, cookies)

        # 页面间隔
        time.sleep(CONFIG["request_delay"] * 1.2)


# 从Excel读取cookies的函数
def read_cookies_from_excel(file_path="cookie.xlsx"):
    """从Excel文件读取cookie列数据"""
    try:
        if not os.path.exists(file_path):
            print(f"[错误] Excel文件 {file_path} 不存在")
            return []

        workbook = openpyxl.load_workbook(file_path)
        # 首先尝试打开名为"cookie"的工作表
        if "cookie" in workbook.sheetnames:
            sheet = workbook["cookie"]
            print("[提示] 使用'cookie'工作表")
        else:
            # 尝试打开包含"cookie"关键字的工作表
            cookie_sheet = None
            for sheet_name in workbook.sheetnames:
                if "cookie" in sheet_name.lower():
                    cookie_sheet = sheet_name
                    break
            if cookie_sheet:
                sheet = workbook[cookie_sheet]
                print(f"[提示] 使用'{cookie_sheet}'工作表")
            else:
                # 使用活动工作表作为备选
                sheet = workbook.active
                print("[提示] 使用活动工作表")

        # 查找cookie列（表头包含"活动cookie"的列）
        cookie_col = None
        for col in range(1, sheet.max_column + 1):
            header = str(sheet.cell(row=1, column=col).value).lower() if sheet.cell(row=1, column=col).value else ""
            if "活动cookie" in header:
                cookie_col = col
                print("[提示] 使用'活动cookie'列")
                break

        if not cookie_col:
            # 尝试查找其他可能的cookie列
            for col in range(1, sheet.max_column + 1):
                header = str(sheet.cell(row=1, column=col).value).lower() if sheet.cell(row=1, column=col).value else ""
                if "cookie" in header:
                    cookie_col = col
                    print(f"[提示] 未找到'活动cookie'列，使用'cookie'列")
                    break

        if not cookie_col:
            # 尝试查找其他可能的cookie相关列
            for col in range(1, sheet.max_column + 1):
                header = str(sheet.cell(row=1, column=col).value).lower() if sheet.cell(row=1, column=col).value else ""
                if "活动" in header:
                    cookie_col = col
                    print(f"[提示] 未找到'活动cookie'或'cookie'列，使用'{header}'列")
                    break

        if not cookie_col:
            print("[错误] Excel文件中未找到包含'活动cookie'、'cookie'或'活动'的列")
            return []

        # 读取所有cookie值（跳过表头）
        cookies_list = []
        for row in range(2, sheet.max_row + 1):
            cookie_value = sheet.cell(row=row, column=cookie_col).value
            if cookie_value and str(cookie_value).strip():
                cookies_list.append(str(cookie_value).strip())

        print(f"[成功] 从Excel文件读取到 {len(cookies_list)} 个cookie")
        return cookies_list

    except Exception as e:
        print(f"[读取Excel失败] {str(e)}")
        return []
